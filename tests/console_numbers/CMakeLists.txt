cmake_minimum_required(VERSION 3.24)
project(UmicomKernelConsoleNumbers LANGUAGES C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
add_executable(umicom-console-number-tests test_numbers.c ../../kernel/console_line.c)
target_include_directories(umicom-console-number-tests PRIVATE ../../include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(umicom-console-number-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
if(BUILD_TESTING)
    foreach(_case IN ITEMS unterminated prefix boundary invalid embedded-nul legacy)
        add_test(NAME "kernel.console.numbers.${_case}" COMMAND umicom-console-number-tests "${_case}")
        set_tests_properties("kernel.console.numbers.${_case}" PROPERTIES TIMEOUT 10)
    endforeach()
endif()
